Available APIs

				

					
						

							Method
							Route
							Purpose
						

					
					
						

							GET
							/api/v1/endpoints
							List known endpoint UUIDs and operating systems.
						

						

							GET
							/api/v1/summary/{endpoint_id}
							Summarize relational events for one endpoint.
						

						

							GET
							/api/v1/alerts
							Return the ten most recent alerts.
						

						

							GET
							/api/v1/alerts?endpoint_id=&min_score=
							Filter alerts by endpoint, minimum score, or both.
